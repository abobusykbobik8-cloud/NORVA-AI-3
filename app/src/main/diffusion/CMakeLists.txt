cmake_minimum_required(VERSION 3.22)
project(norva_media LANGUAGES C CXX)
set(SD_BUILD_EXAMPLES OFF CACHE BOOL "")
set(SD_BUILD_SHARED_LIBS OFF CACHE BOOL "")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "")
set(GGML_NATIVE OFF CACHE BOOL "")
set(GGML_OPENMP OFF CACHE BOOL "")
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "")
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/stable-diffusion.cpp" sdcpp)
add_executable(norva_media main.cpp)
target_link_libraries(norva_media PRIVATE stable-diffusion)
if(ANDROID)
    target_link_libraries(norva_media PRIVATE log)
    target_link_options(norva_media PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-s")
endif()
# Android extracts lib*.so into its executable, read-only native library directory.
set_target_properties(norva_media PROPERTIES OUTPUT_NAME libnorva_diffusion.so RUNTIME_OUTPUT_DIRECTORY "${NORVA_BIN_DIR}")
